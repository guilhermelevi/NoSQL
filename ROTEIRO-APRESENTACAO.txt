1 - entra na pasta do projeto
cd "$HOME/Documents/CEUB/5 Semestre/NoSQL"

2 - sobe Elasticsearch, Kibana e Logstash. Tem que ser o primeiro, para o Logstash ja estar escutando na 12201 quando o Mongo subir
cd lab02-elastic && docker compose up -d

3 - sobe Redis e RedisInsight
cd ../lab03-redis && docker compose up -d

4 - sobe o MongoDB
cd ../lab01-mongodb && docker compose up -d

5 - volta para a raiz do projeto
cd ..

6 - lista os containers rodando, devem aparecer 6
docker ps

7 - confirma o Elasticsearch, devolve a versao e o nome do cluster
curl localhost:9200

8 - estado do cluster. Fica yellow porque com 1 no so as replicas nao tem onde ser alocadas
curl "localhost:9200/_cluster/health?pretty"

9 - confirma o Redis, devolve PONG
docker exec redis redis-cli PING

10 - confirma o MongoDB, devolve ok 1
docker exec mongodb-container mongosh --quiet -u admin -p 123456 --authenticationDatabase admin --eval 'db.adminCommand("ping")'

11 - abre o Kibana
http://localhost:5601

12 - abre o RedisInsight, clicar em Redis Lab 3
http://localhost:5540

13 - string de conexao do MongoDB para a extensao do VS Code
mongodb://admin:123456@localhost:27017/?authSource=admin

14 - mostra o pipeline: recebe GELF na 12201 e indexa no Elasticsearch num indice por dia
cat lab02-elastic/logstash/pipeline/logstash.conf

15 - mostra o outro lado, o driver de log gelf no compose do MongoDB
grep -A4 logging lab01-mongodb/docker-compose.yml

16 - entra no shell do MongoDB
docker exec -it mongodb-container mongosh -u admin -p 123456 --authenticationDatabase admin

17 - muda para o banco demo
use demo

18 - faz o Mongo registrar toda operacao no log. Sem isso ele so registra as acima de 100ms e o insert nao gera linha nenhuma. Some a cada restart
db.setProfilingLevel(0, { slowms: 0 })

19 - insere o documento. Devolve acknowledged true e o insertedId
db.pedidos.insertOne({ cliente: "PROFESSOR-RAUL", valor: 1234.56 })

20 - mostra o documento gravado no MongoDB, comprovando a insercao
db.pedidos.find({ cliente: "PROFESSOR-RAUL" })

21 - quantos documentos existem na colecao
db.pedidos.countDocuments({})

22 - sai do shell do MongoDB
exit

23 - lista os indices de log, um por dia
curl "localhost:9200/_cat/indices/mongodb-logs-*?v"

24 - busca no Elasticsearch o log do insert. O campo message traz a linha do mongod crua, sem tratamento, com o documento inteiro dentro
curl "localhost:9200/mongodb-logs-*/_search?pretty&size=1" -H 'Content-Type: application/json' -d '{"query":{"match_phrase":{"message":"PROFESSOR-RAUL"}}}'

25 - derruba o Logstash
docker stop logstash

26 - insere com o Logstash fora do ar. O Mongo funciona normal e o log se perde, porque GELF e UDP e nao tem buffer
docker exec -it mongodb-container mongosh -u admin -p 123456 --authenticationDatabase admin --eval 'db.getSiblingDB("demo").pedidos.insertOne({ cliente: "SEM-LOGSTASH" })'

27 - sobe o Logstash de volta
docker start logstash

28 - volta para a pasta do projeto
cd "$HOME/Documents/CEUB/5 Semestre/NoSQL"

29 - derruba o MongoDB
cd lab01-mongodb && docker compose down

30 - derruba Elasticsearch, Kibana e Logstash
cd ../lab02-elastic && docker compose down

31 - derruba Redis e RedisInsight
cd ../lab03-redis && docker compose down

32 - volta para a raiz
cd ..

33 - confirma que nao sobrou container
docker ps
